4. Circuit Schematics & Control Firmware

Below is the micro-controller firmware script used for sensor calibration, safety arming, and high-frequency static test data capture:

main_test_stand_controller.cpp C++17 • Arduino / ESP32 Platform 
#include <Arduino.h>
#include <HX711.h>
#include <SPI.h>
#include <SD.h>

// Pin Definitions
const int LOADCELL_DOUT = 16;
const int LOADCELL_SCK  = 17;
const int PYRO_ARM_PIN  = 22;
const int PYRO_FIRE_PIN = 23;

HX711 scale;
File dataFile;

void setup() {
  Serial.begin(115200);
  pinMode(PYRO_ARM_PIN, INPUT_PULLDOWN);
  pinMode(PYRO_FIRE_PIN, OUTPUT);
  digitalWrite(PYRO_FIRE_PIN, LOW);

  scale.begin(LOADCELL_DOUT, LOADCELL_SCK);
  scale.set_scale(2280.f); // Load cell calibration factor
  scale.tare();

  if (!SD.begin(5)) {
    Serial.println("[ERROR] SD Card Initialization Failed!");
  }
  Serial.println("[SYSTEM READY] Awaiting Arming Signal...");
}

void loop() {
  if (digitalRead(PYRO_ARM_PIN) == HIGH) {
    float force_N = scale.get_units() * 9.81f;
    uint32_t t_ms = millis();

    // Stream telemetry over serial & log to SD
    Serial.printf("T=%dms, Force=%.2fN\n", t_ms, force_N);
  }
}
